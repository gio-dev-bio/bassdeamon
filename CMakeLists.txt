cmake_minimum_required(VERSION 3.22)
project(BassDeamon VERSION 0.2.9 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BASSDEAMON_BUILD_PLUGIN "Build the JUCE VST3" ON)
include(CTest)
option(BASSDEAMON_BUILD_BENCHMARKS "Build reproducible DSP benchmarks" OFF)
add_library(bassdeamon_dsp INTERFACE)
target_include_directories(bassdeamon_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
if(BASSDEAMON_BUILD_BENCHMARKS)
    add_executable(dsp_benchmark tests/Benchmark.cpp)
    target_link_libraries(dsp_benchmark PRIVATE bassdeamon_dsp)
endif()
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(manual_attenuation tests/ManualAttenuation.cpp)
    target_link_libraries(manual_attenuation PRIVATE bassdeamon_dsp)
    add_test(NAME calibrated_manual_attenuation COMMAND manual_attenuation)
    add_executable(analyzer_response tests/AnalyzerResponse.cpp)
    target_link_libraries(analyzer_response PRIVATE bassdeamon_dsp)
    add_test(NAME analyzer_residual_and_rate COMMAND analyzer_response)
    add_executable(analyzer_tests tests/AnalyzerTests.cpp)
    target_link_libraries(analyzer_tests PRIVATE bassdeamon_dsp)
    add_test(NAME analyzer_precision_contracts COMMAND analyzer_tests)
    add_executable(analyzer_concurrency tests/AnalyzerConcurrency.cpp)
    target_link_libraries(analyzer_concurrency PRIVATE bassdeamon_dsp Threads::Threads)
    add_test(NAME analyzer_bounded_consumer COMMAND analyzer_concurrency)
    set_tests_properties(analyzer_bounded_consumer PROPERTIES TIMEOUT 15)
    add_executable(analyzer_matrix tests/AnalyzerMatrix.cpp)
    target_link_libraries(analyzer_matrix PRIVATE bassdeamon_dsp)
    add_test(NAME analyzer_audio_matrix COMMAND analyzer_matrix ${CMAKE_CURRENT_BINARY_DIR}/analyzer-matrix.csv)
    add_executable(usability_tests tests/UsabilityTests.cpp)
    target_link_libraries(usability_tests PRIVATE bassdeamon_dsp)
    add_test(NAME usability_contracts COMMAND usability_tests)
    add_executable(new_feature_tests tests/NewFeatureTests.cpp)
    target_link_libraries(new_feature_tests PRIVATE bassdeamon_dsp Threads::Threads)
    add_test(NAME softtube_analyzer_contracts COMMAND new_feature_tests)
    add_executable(dsp_tests tests/DspTests.cpp)
    target_link_libraries(dsp_tests PRIVATE bassdeamon_dsp)
    add_test(NAME dsp_contracts COMMAND dsp_tests)
    add_executable(signal_tests tests/SignalTests.cpp)
    target_link_libraries(signal_tests PRIVATE bassdeamon_dsp)
    add_test(NAME signal_contracts COMMAND signal_tests ${CMAKE_CURRENT_BINARY_DIR}/signal-results)
endif()
if(BASSDEAMON_BUILD_PLUGIN)
    include(FetchContent)
    # Reproducible JUCE 9.0.5 revision; optional local checkout for offline builds.
    set(BASSDEAMON_JUCE_PATH "" CACHE PATH "Existing JUCE checkout")
    if(BASSDEAMON_JUCE_PATH)
        add_subdirectory(${BASSDEAMON_JUCE_PATH} ${CMAKE_BINARY_DIR}/JUCE)
    else()
        FetchContent_Declare(JUCE
            URL https://codeload.github.com/juce-framework/JUCE/tar.gz/d60b7a78030661574a615b586d9325ba7a7fa7dd
            URL_HASH SHA256=8237b2976871982eb9107fd67b576d95bf02355d99540a0127c4e7c001ecd9f5)
        FetchContent_MakeAvailable(JUCE)
    endif()
    juce_add_plugin(BassDeamon
        COMPANY_NAME "Guy_BioRg" BUNDLE_ID "audio.guybiorg.bassdeamon"
        PLUGIN_MANUFACTURER_CODE Gbio PLUGIN_CODE BsDm
        FORMATS VST3 PRODUCT_NAME "BassDeamon"
        IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
        COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics)
    target_sources(BassDeamon PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
    target_compile_definitions(BassDeamon PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(BassDeamon PRIVATE bassdeamon_dsp juce::juce_audio_utils
        PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
    if(BUILD_TESTING)
        juce_add_console_app(plugin_tests PRODUCT_NAME "BassDeamon plugin contracts")
        target_sources(plugin_tests PRIVATE tests/PluginTests.cpp tests/GuiCapture.cpp tests/PluginSignalTests.cpp tests/GuiInteractionTests.cpp)
        set_target_properties(plugin_tests PROPERTIES OUTPUT_NAME plugin_tests)
        target_include_directories(plugin_tests PRIVATE Source)
        target_link_libraries(plugin_tests PRIVATE BassDeamon juce::juce_audio_utils)
        add_test(NAME plugin_contracts COMMAND plugin_tests)
    endif()
endif()
